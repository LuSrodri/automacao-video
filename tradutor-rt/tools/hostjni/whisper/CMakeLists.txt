cmake_minimum_required(VERSION 3.22)
project(hostjni_whisper C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(JNI REQUIRED)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
foreach(o WHISPER_BUILD_TESTS WHISPER_BUILD_EXAMPLES WHISPER_BUILD_SERVER WHISPER_SDL2 WHISPER_CURL)
  set(${o} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(${ROOT}/third_party/whisper.cpp whisper-build)
add_library(whisper_jni SHARED ${ROOT}/whisper-jni/src/main/cpp/whisper_jni.cpp)
target_include_directories(whisper_jni PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/.. ${JNI_INCLUDE_DIRS})
target_link_libraries(whisper_jni PRIVATE whisper)
target_link_options(whisper_jni PRIVATE -Wl,--exclude-libs,ALL)
set_target_properties(whisper_jni PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${OUT_DIR})
